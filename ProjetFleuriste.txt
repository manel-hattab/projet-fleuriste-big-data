// A1 — Chiffre d'affaires total par catégorie

  { $unwind: "$lignes" },
  {
    $lookup: {
      from: "produits",
      localField: "lignes.produit_id",
      foreignField: "_id",
      as: "produit_info"
    }
  },
  { $unwind: "$produit_info" },
  {
    $lookup: {
      from: "categories",
      localField: "produit_info.categorie_id",
      foreignField: "_id",
      as: "categorie_info"
    }
  },
  { $unwind: "$categorie_info" },
  {
    $group: {
      _id: "$categorie_info.nom",
      chiffre_affaires: {
        $sum: { $multiply: ["$lignes.quantite", "$lignes.prix_unitaire"] }
      },
      nb_articles_vendus: { $sum: "$lignes.quantite" }
    }
  },
  { $sort: { chiffre_affaires: -1 } }
]);
